# Verified working combination (2026-09-05, CPU only):
#
#   conda create -n toxocr-env python=3.10
#   pip install torch==1.13.1 --index-url https://download.pytorch.org/whl/cpu
#   pip install -r toxocr/requirements.txt
#   pip install "numpy<2"   # molscribe's pinned deps (timm, torchvision 0.14,
#                           # albumentations 1.1.0) predate NumPy 2.0's ABI
#                           # break; installing them alone resolves numpy 2.x
#                           # and several import warnings turn into failures.
#
